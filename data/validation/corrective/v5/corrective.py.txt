"""Heurísticas versionadas de release corretiva (#58), sem acesso a votos humanos."""

from __future__ import annotations

import re
from dataclasses import asdict, dataclass

# Não se usa substring: 'prefix', 'fixture' e 'fixed-point' não são um 'fix'.
CORRECTION = re.compile(r"\b(?:fix(?:es|ed|ing)?|bugfix(?:es)?|hotfix(?:es)?|revert(?:s|ed|ing)?)\b(?!-point)", re.I)
FEATURE = re.compile(r"\b(?:feat(?:ure)?s?|enhancement(?:s)?|breaking changes?)\b", re.I)
MAINTENANCE = re.compile(r"^(?:docs?|tests?|ci|build|chore|style)(?:\([^)]*\))?!?:", re.I)
MAINTENANCE_FIX = re.compile(r"^fix\((?:docs?|tests?|ci|build|deps?|style)\)!?:", re.I)
SEMVER = re.compile(r"v?(0|[1-9]\d*)\.(0|[1-9]\d*)\.(0|[1-9]\d*)(?:\+[0-9A-Za-z-]+(?:\.[0-9A-Za-z-]+)*)?")
FEATURE_ACTION = re.compile(r"\b(?:add(?:s|ed|ing)?|introduc(?:e|es|ed|ing)|new|support(?:s|ed|ing)?|improvements?)\b", re.I)
POLICIES = {
    "v1": "Incremento apenas do patch SemVer OU mensagem de commit com fix, bugfix, hotfix ou revert.",
    "v2": "Predominância de correções sobre features nas notas; na ausência de sinais nas notas, nos commits. "
          "Ignora commits de manutenção explicitamente tipados; empate é negativo. "
          "Patch SemVer é fallback somente quando não há sinais de correção/feature.",
    "v3": "Maioria estrita de entradas corretivas entre TODAS as mudanças nas notas; "
          "listas de contribuidores, links de changelog e blocos de código não são mudanças. "
          "Sem entradas nas notas, usa maioria estrita de assuntos de commits não tipados como manutenção/merge. "
          "Notas com mudanças têm precedência; patch SemVer isolado não é evidência suficiente.",
    "v4": "Mesma maioria estrita de v3 entre mudanças funcionais: exclui seções de dependências, "
          "documentação e mudanças internas, assuntos explicitamente tipados como manutenção/refactor "
          "e entradas administrativas de release. Remove prefixos de SHA antes de identificar o tipo. "
          "Mantém mudanças funcionais desconhecidas no denominador e não usa patch isolado.",
    "v5": "Refinamento lexical de v2: reconhece adições, introduções, suporte e melhorias como "
          "sinais funcionais concorrentes às correções; ignora manutenção explicitamente tipada "
          "e seções administrativas. Compara correções contra sinais funcionais nas notas; "
          "na ausência de sinais usa commits. Empate é negativo; patch isolado não classifica.",
}


def semver(tag: str) -> tuple[int, int, int] | None:
    match = SEMVER.fullmatch(tag.strip())
    return tuple(map(int, match.groups())) if match else None


def patch_only(previous_tag: str, tag: str) -> bool:
    previous, current = semver(previous_tag), semver(tag)
    return bool(previous and current and previous[:2] == current[:2] and current[2] > previous[2])


def message_signals(message: str, *, exclude_maintenance: bool = False) -> tuple[bool, bool]:
    subject = message.splitlines()[0] if message else ""
    if exclude_maintenance and (MAINTENANCE.match(subject) or MAINTENANCE_FIX.match(subject)):
        return False, False
    return bool(CORRECTION.search(subject)), bool(FEATURE.search(subject))


def note_signals(body: str) -> tuple[int, int]:
    """Conta entradas, não palavras; seções Markdown propagam o tipo às listas."""
    fixes = features = 0
    section = (False, False)
    for line in body.splitlines():
        text = line.strip()
        if not text:
            continue
        if text.startswith("#"):
            section = message_signals(text)
            continue
        if text.startswith(("- ", "* ", "+ ")):
            own = message_signals(text)
            signals = own if any(own) else section
        else:
            signals = message_signals(text)
        fixes += int(signals[0])
        features += int(signals[1])
    return fixes, features


@dataclass(frozen=True)
class Prediction:
    version: str
    corrective_release: str
    reason: str
    patch_only: bool
    correction_commits: int
    feature_commits: int
    correction_notes: int
    feature_notes: int
    commit_context_complete: bool
    note_change_entries: int = 0
    commit_change_entries: int = 0

    def to_dict(self) -> dict:
        return asdict(self)


def change_entries(body: str, *, ignore_maintenance: bool = False) -> list[bool]:
    """Entradas Markdown de mudança; desconhecidas contam no denominador."""
    entries, paragraphs = [], []
    section = (False, False)
    skip = fenced = False
    for line in body.splitlines():
        text = line.strip()
        if text.startswith(("```", "~~~")):
            fenced = not fenced
            continue
        if fenced or not text or text.startswith(("<!--", "**Full Changelog**", "Full Changelog", "http://", "https://")):
            continue
        if text.startswith("#"):
            skip = bool(re.search(r"contributors|acknowledg|thanks", text, re.I))
            if ignore_maintenance:
                skip = skip or bool(re.search(r"dependenc|documentation|internal changes|maintenance|refactor|\btests?\b|\bbuild\b|\bci\b", text, re.I))
            section = message_signals(text)
            continue
        if skip:
            continue
        bullet = re.match(r"^(?:[-*+]\s+|\d+[.)]\s+)(.+)", text)
        content = bullet.group(1) if bullet else text
        if ignore_maintenance:
            content = re.sub(r"^[0-9a-f]{7,40}\s+", "", content, flags=re.I)
            if MAINTENANCE.match(content) or MAINTENANCE_FIX.match(content) or re.match(r"refactor(?:\([^)]*\))?!?:|(?:bump|released|release version)\b", content, re.I):
                continue
        own = message_signals(content, exclude_maintenance=True)
        signals = own if any(own) else section
        # Manutenção explicitamente tipada não vira correção por herança de seção.
        correction = signals[0] and not (MAINTENANCE.match(content) or MAINTENANCE_FIX.match(content))
        if bullet:
            entries.append(bool(correction))
        elif any(own):
            paragraphs.append(bool(correction))
    return entries if entries else paragraphs


def functional_signals(text: str) -> tuple[bool, bool]:
    text = re.sub(r"^[0-9a-f]{7,40}\s+", "", text, flags=re.I)
    if MAINTENANCE.match(text) or MAINTENANCE_FIX.match(text) or re.match(r"refactor(?:\([^)]*\))?!?:", text, re.I):
        return False, False
    fix, feature = message_signals(text)
    return fix, not fix and (feature or bool(FEATURE_ACTION.search(text)))


def functional_notes(body: str) -> tuple[int, int]:
    fixes = features = 0
    section = (False, False)
    skip = fenced = False
    for line in body.splitlines():
        text = line.strip()
        if text.startswith(("```", "~~~")):
            fenced = not fenced
            continue
        if fenced or not text or text.startswith(("<!--", "**Full Changelog**", "http://", "https://")):
            continue
        if text.startswith("#"):
            skip = bool(re.search(r"contributors|acknowledg|thanks|dependenc|documentation|internal changes", text, re.I))
            section = functional_signals(text)
            continue
        if skip:
            continue
        bullet = re.match(r"^(?:[-*+]\s+|\d+[.)]\s+)(.+)", text)
        content = bullet.group(1) if bullet else text
        own = functional_signals(content)
        if MAINTENANCE.match(content) or MAINTENANCE_FIX.match(content):
            continue
        signals = own if any(own) else section if bullet else own
        fixes += int(signals[0])
        features += int(signals[1])
    return fixes, features


def predict(*, tag: str, previous_tag: str = "", body: str = "", messages=(),
            interval_status: str = "", version: str = "v1") -> Prediction:
    if version not in POLICIES:
        raise ValueError(f"Versão de heurística desconhecida: {version}")
    patch = patch_only(previous_tag, tag)
    messages = list(messages)
    signals = [message_signals(message, exclude_maintenance=version != "v1") for message in messages]
    fixes, features = sum(s[0] for s in signals), sum(s[1] for s in signals)
    note_fixes, note_features = note_signals(body)
    note_entries, commit_entries = [], []
    if version == "v5":
        note_fixes, note_features = functional_notes(body)
        functional = [functional_signals(m.splitlines()[0]) for m in messages if m.strip() and not m.lower().startswith("merge ")]
        fixes, features = sum(s[0] for s in functional), sum(s[1] for s in functional)
        positive = note_fixes > note_features if note_fixes or note_features else fixes > features
        reason = "functional_notes_balance" if note_fixes or note_features else "functional_commits_balance"
    elif version in ("v3", "v4"):
        note_entries = change_entries(body, ignore_maintenance=version == "v4")
        subjects = [m.splitlines()[0] for m in messages if m.strip()]
        subjects = [s for s in subjects if not MAINTENANCE.match(s) and not MAINTENANCE_FIX.match(s)
                    and not s.lower().startswith("merge ")]
        if version == "v4":
            subjects = [s for s in subjects if not re.match(r"refactor(?:\([^)]*\))?!?:", s, re.I)]
        commit_entries = [message_signals(s)[0] for s in subjects]
        evidence = note_entries if note_entries else commit_entries
        positive = bool(evidence) and sum(evidence) > len(evidence) / 2
        reason = "notes_majority_all_changes" if note_entries else "commits_majority_all_changes" if commit_entries else "no_change_evidence"
    elif version == "v1":
        positive = patch or fixes > 0
        reason = "patch_or_correction_commit" if positive else "no_positive_signal"
    elif note_fixes or note_features:
        positive = note_fixes > note_features
        reason = "notes_corrections_predominate" if positive else "notes_not_predominantly_corrective"
    elif fixes or features:
        positive = fixes > features
        reason = "commits_corrections_predominate" if positive else "commits_not_predominantly_corrective"
    else:
        positive = patch
        reason = "patch_fallback" if positive else "no_positive_signal"
    return Prediction(version, "yes" if positive else "no", reason, patch, fixes, features,
                      note_fixes, note_features, interval_status == "complete", len(note_entries), len(commit_entries))
